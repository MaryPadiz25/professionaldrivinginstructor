// =============================================
// FIRESTORE SECURITY RULES
// Paste this into: Firebase Console → Firestore Database → Rules tab → Publish
// =============================================
//
// What this enforces:
//   - "applications" collection: ANYONE can create a new application
//     (so the public join-form works from any device, no login needed),
//     but NOBODY can read, list, update, or delete them unless they are
//     signed in as the admin. This is what stops a stranger from opening
//     dev tools and reading every applicant's personal details.
//   - "live_profiles" collection: anyone can READ (so the public website
//     can display approved instructor profiles), but only the signed-in
//     admin can write (create/update/delete) — this replaces the old
//     "approve and publish" step.
//
// To make yourself the admin: Firebase Console → Authentication → Users
// → Add user (email + password). Sign in with that exact email on the
// admin page.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /applications/{appId} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }

    match /live_profiles/{profileId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
